**操作步骤**：

5. 复制并粘贴以下完整代码：

```python
import pandas as pd
import json

def parse_excel(file_path):
    """
    解析Excel文件，提取关键数据
    """
    try:
        # 读取Excel文件
        df = pd.read_excel(file_path, sheet_name=0)

        # 提取第一行数据（测评结果）
        # A列（索引0）：序号
        # CX列（索引83）：微信昵称
        # DB列（索引87）：总分
        # DC-DK列（索引87-95）：9个能力维度分数

        data = {
            'sequence_no': str(df.iloc[0, 0]).strip(),  # A列：序号
            'wechat_name': str(df.iloc[0, 83]).strip(),  # CX列：微信昵称（第84列）
            'total_score': float(df.iloc[0, 87]),  # DB列：总分（第88列）
        }

        # 提取9个能力维度分数（DC-DK列，索引87-95）
        abilities = ['执行力', '协调力', '优化力', '统筹力', '预见力', '业务力', '财务力', '领导力', '决策力']
        ability_scores = {}

        for i, ability in enumerate(abilities):
            col_idx = 87 + i  # DC列是第88列（索引87）
            try:
                score = float(df.iloc[0, col_idx])
                ability_scores[ability] = score
            except:
                ability_scores[ability] = 0.0

        data['ability_scores'] = ability_scores

        return {
            'success': True,
            'data': data,
            'message': 'Excel解析成功'
        }

    except Exception as e:
        return {
            'success': False,
            'error': f'Excel解析失败: {str(e)}',
            'data': None
        }

# 执行解析
result = parse_excel(excel_file)

# 返回结果
return result
```

```python
def calculate_rank_and_grades(parsed_data):
    """
    根据分数计算综合段位和各能力等级
    """
    total_score = parsed_data['total_score']
    ability_scores = parsed_data['ability_scores']

    # ========== 1. 计算综合段位 ==========
    if total_score <= 15.09:
        rank = '一段'
        rank_num = 1
    elif total_score <= 30.09:
        rank = '二段'
        rank_num = 2
    elif total_score <= 40.09:
        rank = '三段'
        rank_num = 3
    elif total_score <= 45.09:
        rank = '四段'
        rank_num = 4
    elif total_score <= 55.09:
        rank = '五段'
        rank_num = 5
    elif total_score <= 70.09:
        rank = '六段'
        rank_num = 6
    elif total_score <= 75.09:
        rank = '七段'
        rank_num = 7
    elif total_score <= 85.09:
        rank = '八段'
        rank_num = 8
    else:
        rank = '九段'
        rank_num = 9

    # ========== 2. 计算各能力等级 ==========
    def get_ability_grade(ability, score):
        """
        根据能力类型和分数确定等级
        不同能力维度使用不同的评分标准
        """
        if ability in ['执行力', '协调力', '优化力']:
            # 核心基石：8分制
            if score >= 7.2: return 'A'
            elif score >= 6.4: return 'B'
            elif score >= 5.6: return 'C'
            elif score >= 4.8: return 'D'
            else: return 'E'

        elif ability in ['统筹力', '预见力', '业务力']:
            # 价值引擎：10分制
            if score >= 9.0: return 'A'
            elif score >= 8.0: return 'B'
            elif score >= 7.0: return 'C'
            elif score >= 6.0: return 'D'
            else: return 'E'

        else:  # 财务力、领导力、决策力
            # 领导潜能：12分制
            if score >= 10.8: return 'A'
            elif score >= 9.6: return 'B'
            elif score >= 8.4: return 'C'
            elif score >= 7.2: return 'D'
            else: return 'E'

    # 计算所有能力的等级
    ability_grades = {}
    for ability, score in ability_scores.items():
        grade = get_ability_grade(ability, score)
        ability_grades[ability] = {
            'score': score,
            'grade': grade
        }

    return {
        'rank': rank,
        'rank_num': rank_num,
        'total_score': total_score,
        'ability_grades': ability_grades,
        'sequence_no': parsed_data['sequence_no'],
        'wechat_name': parsed_data['wechat_name'],
        'message': f'段位计算成功：{rank}'
    }

# 执行计算
result = calculate_rank_and_grades(parsed_data)

# 返回结果
return result
```

4. 配置输入变量：`parsed_data`（Object类型）
5. 配置输出变量：`rank_data`（Object类型）
6. 保存
